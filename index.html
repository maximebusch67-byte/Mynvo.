// ======================================== // SCRIPT DE CONTRÔLES PERSONNALISÉS // Pour Subway Surfers - Version Mynvo // ======================================== (function() { 'use strict'; // Configuration des touches par défaut let keyMapping = { up: 'Z', down: 'S', left: 'Q', right: 'D', jump: ' ' }; // Correspondance touches -> keycodes des flèches const arrowKeyCodes = { up: 38, // ArrowUp down: 40, // ArrowDown left: 37, // ArrowLeft right: 39, // ArrowRight jump: 32 // Space }; // Variable pour savoir si on est en train de changer une touche let isChangingKey = false; // Charger la configuration sauvegardée function loadKeyMapping() { try { const saved = localStorage.getItem('mynvoKeyMapping'); if (saved) { keyMapping = JSON.parse(saved); console.log('✅ Configuration des touches chargée:', keyMapping); } } catch (e) { console.warn('⚠️ Impossible de charger la configuration des touches'); } } // Sauvegarder la configuration function saveKeyMapping() { try { localStorage.setItem('mynvoKeyMapping', JSON.stringify(keyMapping)); console.log('💾 Configuration des touches sauvegardée'); } catch (e) { console.warn('⚠️ Impossible de sauvegarder la configuration'); } } // Créer un événement clavier simulé function createKeyEvent(type, keyCode) { const keyNames = { 38: 'ArrowUp', 40: 'ArrowDown', 37: 'ArrowLeft', 39: 'ArrowRight', 32: ' ' }; const keyCodes = { 38: 'ArrowUp', 40: 'ArrowDown', 37: 'ArrowLeft', 39: 'ArrowRight', 32: 'Space' }; return new KeyboardEvent(type, { key: keyNames[keyCode], code: keyCodes[keyCode], keyCode: keyCode, which: keyCode, bubbles: true, cancelable: true, composed: true, view: window }); } // Gestionnaire principal des touches function handleKeyEvent(e) { // Ignorer si on est dans un input ou si on change une touche if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA' || isChangingKey) { return; } const pressedKey = e.key.toUpperCase(); let mappedKeyCode = null; // Vérifier si la touche pressée correspond à une de nos touches mappées for (const [action, key] of Object.entries(keyMapping)) { if (key.toUpperCase() === pressedKey) { mappedKeyCode = arrowKeyCodes[action]; break; } } // Si une correspondance est trouvée if (mappedKeyCode) { e.preventDefault(); e.stopPropagation(); e.stopImmediatePropagation(); // Créer et dispatcher l'événement simulé const simulatedEvent = createKeyEvent(e.type, mappedKeyCode); // Essayer d'injecter dans l'iframe const iframe = document.getElementById('game-iframe'); if (iframe) { try { // Tenter l'injection dans l'iframe (peut échouer avec cross-origin) if (iframe.contentWindow) { iframe.contentWindow.dispatchEvent(simulatedEvent); } if (iframe.contentDocument) { iframe.contentDocument.dispatchEvent(simulatedEvent); // Cibler spécifiquement le canvas const canvas = iframe.contentDocument.querySelector('canvas'); if (canvas) { canvas.dispatchEvent(simulatedEvent); } } } catch (err) { // En cas d'erreur cross-origin, dispatcher sur window window.dispatchEvent(simulatedEvent); document.dispatchEvent(simulatedEvent); } } // Dispatcher aussi sur le document principal document.dispatchEvent(simulatedEvent); window.dispatchEvent(simulatedEvent); // Feedback visuel (optionnel) showKeyPress(pressedKey, mappedKeyCode); } } // Feedback visuel pour le debug function showKeyPress(originalKey, mappedKeyCode) { const keyNames = { 38: '↑', 40: '↓', 37: '←', 39: '→', 32: 'SPACE' }; console.log(`🎮 ${originalKey} → ${keyNames[mappedKeyCode]}`); } // Initialiser les écouteurs d'événements function initializeListeners() { // Retirer les anciens écouteurs s'ils existent if (window.customKeyHandler) { ['keydown', 'keyup'].forEach(type => { document.removeEventListener(type, window.customKeyHandler, true); window.removeEventListener(type, window.customKeyHandler, true); }); } // Créer le nouveau gestionnaire const handler = handleKeyEvent.bind(this); window.customKeyHandler = handler; // Ajouter les écouteurs avec capture pour intercepter avant le jeu ['keydown', 'keyup'].forEach(type => { document.addEventListener(type, handler, { capture: true, passive: false }); window.addEventListener(type, handler, { capture: true, passive: false }); }); console.log('🎮 Écouteurs d\'événements initialisés'); } // Fonction pour changer une touche (à appeler depuis l'interface) window.updateKeyMapping = function(action, newKey) { if (keyMapping.hasOwnProperty(action)) { keyMapping[action] = newKey.toUpperCase(); saveKeyMapping(); console.log(`✅ Touche ${action} changée pour: ${newKey}`); return true; } return false; }; // Fonction pour réinitialiser les touches par défaut window.resetKeyMapping = function() { keyMapping = { up: 'Z', down: 'S', left: 'Q', right: 'D', jump: ' ' }; saveKeyMapping(); console.log('🔄 Touches réinitialisées aux valeurs par défaut'); }; // Fonction pour obtenir la configuration actuelle window.getKeyMapping = function() { return { ...keyMapping }; }; // Fonction pour indiquer qu'on change une touche window.setChangingKey = function(isChanging) { isChangingKey = isChanging; }; // Attendre que le DOM soit prêt if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', function() { loadKeyMapping(); initializeListeners(); }); } else { loadKeyMapping(); initializeListeners(); } console.log('✅ Script de contrôles personnalisés chargé!'); console.log('📋 Configuration actuelle:', keyMapping); })(); // ======================================== // INTÉGRATION AVEC L'INTERFACE EXISTANTE // ======================================== // Fonction pour mettre à jour l'interface avec les touches actuelles function syncUIWithKeyMapping() { const mapping = window.getKeyMapping(); if (mapping) { document.getElementById('key-up').value = mapping.up; document.getElementById('key-down').value = mapping.down; document.getElementById('key-left').value = mapping.left; document.getElementById('key-right').value = mapping.right; document.getElementById('key-jump').value = mapping.jump; } } // Modifier la fonction changeKey existante function changeKey(action) { window.setChangingKey(true); const input = document.getElementById(`key-${action}`); input.style.borderColor = 'var(--warning)'; input.placeholder = 'Appuyez sur une touche...'; input.value = ''; // Créer un écouteur temporaire pour capturer la nouvelle touche const captureKey = (e) => { e.preventDefault(); e.stopPropagation(); const newKey = e.key === ' ' ? ' ' : e.key.toUpperCase(); // Vérifier si la touche n'est pas déjà utilisée const currentMapping = window.getKeyMapping(); let alreadyUsed = false; for (const [existingAction, existingKey] of Object.entries(currentMapping)) { if (existingKey === newKey && existingAction !== action) { alert(`❌ Cette touche est déjà utilisée pour: ${existingAction}`); alreadyUsed = true; break; } } if (!alreadyUsed) { // Mettre à jour la touche window.updateKeyMapping(action, newKey); input.value = newKey; input.style.borderColor = 'var(--success)'; // Feedback visuel setTimeout(() => { input.style.borderColor = 'var(--primary)'; }, 500); } else { // Restaurer la valeur précédente input.value = currentMapping[action]; input.style.borderColor = 'var(--danger)'; setTimeout(() => { input.style.borderColor = 'var(--primary)'; }, 500); } input.placeholder = ''; window.setChangingKey(false); // Retirer l'écouteur document.removeEventListener('keydown', captureKey, true); }; document.addEventListener('keydown', captureKey, { capture: true, once: false }); // Annuler après 5 secondes setTimeout(() => { if (input.value === '') { const currentMapping = window.getKeyMapping(); input.value = currentMapping[action]; input.style.borderColor = 'var(--primary)'; input.placeholder = ''; window.setChangingKey(false); document.removeEventListener('keydown', captureKey, true); } }, 5000); } // Modifier la fonction resetControls existante function resetControls() { window.resetKeyMapping(); syncUIWithKeyMapping(); // Feedback visuel const inputs = document.querySelectorAll('.key-input'); inputs.forEach(input => { input.style.borderColor = 'var(--success)'; setTimeout(() => { input.style.borderColor = 'var(--primary)'; }, 500); }); } // Synchroniser au chargement document.addEventListener('DOMContentLoaded', function() { setTimeout(syncUIWithKeyMapping, 100); });
